# Laziness
Brei executes workflows in Asyncio, through lazy evaluation and memoization. The first time a `Lazy` task is requested, the `LazyDB` starts it as an `asyncio.Task` and stores that future on the task object. When multiple tasks ask for the result of the same dependent task, they all await the same future, so the computation is performed only once.

A task waiting for one of its dependencies is recorded in its `_awaiting` list. Before a task starts waiting on a dependency, we check that the dependency isn't (indirectly) waiting on the task already. Because asyncio is single threaded, the task that would close a cycle always finds it, no matter in which order the tasks were started.

## Results
One common problem when using workflow systems in Python, is that errors are not tracable to their source. It is quite common to get a stack-trace that leads to the internals of the workflow system, but not to the cause of the error. The way around this, is not to use Python's exception system (at least not in the outer layer), but rather signify errors by returning `Failure` objects.

``` {.python file=test/test_result.py}
from brei.result import Failure, TaskFailure, Ok
from hypothesis import given
from hypothesis.strategies import builds, booleans, text, integers


results = builds(lambda b, t, i: Ok(i) if b else TaskFailure(t),
                 booleans(), text(min_size=1), integers())


@given(results)
def test_result(r):
    assert (r and hasattr(r, "value")) or (not r and isinstance(r, Failure))
```

``` {.python file=brei/result.py}
from typing import TypeVar, Generic
from dataclasses import dataclass


T = TypeVar("T")
R = TypeVar("R")


class Failure(Exception):
    def __bool__(self):
        return False


@dataclass
class MissingFailure(Failure, Generic[T]):
    target: T

    def __str__(self):
        return f"Missing dependency: {self.target}"


@dataclass
class TaskFailure(Failure):
    message: str

    def __post_init__(self):
        Exception.__init__(self, self.message)


@dataclass
class DependencyFailure(Failure, Generic[T]):
    dependencies: dict[T, Failure]

    def __str__(self):
        return "\n".join(f"{key} -> {fail}" for key, fail in self.dependencies.items())

@dataclass
class Ok(Generic[R]):
    value: R

    def __bool__(self):
        return True


Result = Failure | Ok[R]
```

## Lazy evaluation

``` {.python file=brei/lazy.py}
from __future__ import annotations
from dataclasses import dataclass, field
from typing import Generic, Optional, TypeVar, Any
import asyncio

from .errors import CyclicWorkflowError
from .logging import logger
from .result import Failure, Result, Ok, DependencyFailure, MissingFailure

T = TypeVar("T")
R = TypeVar("R")

log = logger()


@dataclass
class Phony:
    name: str

    @classmethod
    def from_str(cls, s: str) -> Phony:
        if s[0] == "#":
            return Phony(s[1:])
        raise ValueError("A phony target should start with a `#` character.")

    def __str__(self):
        return f"#{self.name}"

    def __hash__(self):
        return hash(str(self))


@dataclass
class Lazy(Generic[T, R]):
    """Base class for tasks that are tagged with type `T` (usually `str` or
    `Path`) and representing values of type `R`.

    To implement a specific task, you need to implement the asynchronous
    `run` method, which should return a value of `R` or throw `TaskFailure`.
    If `run` returns another `Lazy` object, that is run in turn, and its
    result is the result of this task.

    Attributes:
        creates: list of target identifiers, for instance paths that are
            generated by running a particular task.
        requires: list of dependency identifiers. All of these need to
            be realized before the task can run.
        result (property): value of the result, once the task was run. This
            throws an exception if accessed before the task is complete.
    """

    creates: list[T]
    requires: list[T]

    _future: Optional[asyncio.Future[R]] = field(default=None, init=False)
    _awaiting: list[Lazy] = field(default_factory=list, init=False)

    def __bool__(self):
        return self._future is not None and self._future.done() \
            and not self._future.cancelled() and self._future.exception() is None

    @property
    def result(self) -> R:
        if self._future is None or not self._future.done():
            raise ValueError("Task has not run yet.")
        if not self:
            raise ValueError("Task has failed.")
        return self._future.result()

    async def run(self, *, db) -> R:
        raise NotImplementedError()

    def reset(self):
        self._future = None
        self._awaiting = []


TaskT = TypeVar("TaskT", bound=Lazy)


class MissingDependency(Exception):
    pass


@dataclass
class LazyDB(Generic[T, TaskT]):
    """Collect tasks and coordinate running a task from a task identifier."""

    tasks: list[TaskT] = field(default_factory=list)
    index: dict[T, TaskT] = field(default_factory=dict)

    async def run(self, t: T, **kwargs) -> Result[Any]:
        """Run the task that creates `t`, if it hasn't run yet. Keyword
        arguments are passed on to `Lazy.run`. Returns `Ok` with the value of
        the task, or a `Failure`."""
        try:
            return Ok(await self._start(self._get_task(t), **kwargs))
        except Failure as f:
            return f

    def _get_task(self, t: T) -> Lazy:
        if t in self.index:
            return self.index[t]
        try:
            return self.on_missing(t)
        except MissingDependency:
            raise MissingFailure(t)

    def _start(self, task: Lazy, **kwargs) -> asyncio.Future:
        if task._future is None:
            task._future = asyncio.ensure_future(self._run(task, **kwargs))
        return task._future

    async def _run(self, task: Lazy, **kwargs) -> Any:
        current = task
        while True:
            await self._run_deps(task, current.requires, **kwargs)
            value = await current.run(**kwargs)
            if not isinstance(value, Lazy):
                return value
            current = value

    async def _run_deps(self, waiter: Lazy, requires: list[T], **kwargs):
        """Run the dependencies of `waiter`. Failures are collected into a
        `DependencyFailure`, other exceptions are passed on."""
        def start(t: T) -> asyncio.Future:
            try:
                dep = self._get_task(t)
            except MissingFailure as f:
                future = asyncio.get_running_loop().create_future()
                future.set_exception(f)
                return future
            self._check_cycle(waiter, dep)
            waiter._awaiting.append(dep)
            return self._start(dep, **kwargs)

        results = await asyncio.gather(*map(start, requires), return_exceptions=True)
        for r in results:
            if isinstance(r, BaseException) and not isinstance(r, Failure):
                raise r
        failed = {t: r for t, r in zip(requires, results) if isinstance(r, Failure)}
        if failed:
            raise DependencyFailure(failed)

    def _check_cycle(self, waiter: Lazy, dep: Lazy):
        """Raise `CyclicWorkflowError` if `dep` is (indirectly) waiting on `waiter`."""
        paths = [[dep]]
        seen: set[int] = set()
        while paths:
            path = paths.pop()
            if path[-1] is waiter:
                raise CyclicWorkflowError([t.creates[0] for t in [waiter, *path] if t.creates])
            if id(path[-1]) in seen:
                continue
            seen.add(id(path[-1]))
            paths.extend(path + [d] for d in path[-1]._awaiting)

    def on_missing(self, t: T) -> TaskT:
        raise MissingDependency()

    def add(self, task: TaskT):
        """Add a task to the DB."""
        log.debug(f"adding task ===\n{task}")
        self.tasks.append(task)
        for target in task.creates:
            self.index[target] = task

    def clean(self):
        self.tasks = []
        self.index = {}

    def reset(self):
        for t in self.tasks:
            t.reset()
```
